### 词法分析核心 — DFA 逐行扫描

```java
// src/lexer/Lexer.java
public class Lexer {
    // PL/0 全部关键字
    static final Set<String> KW = new HashSet<>(Arrays.asList(
        "const", "var", "procedure", "call", "begin", "end",
        "if", "then", "else", "while", "do", "read", "write", "odd"
    ));

    public List<Token> tokenize(String code) {
        toks.clear();
        errs.clear();
        String[] lines = code.split("\n");
        boolean inBlock = false;

        for (int i = 0; i < lines.length; i++) {
            String line = lines[i];
            int lineno = i + 1;
            int col = 0;

            // 处理多行注释 /* ... */
            if (inBlock) {
                int e = line.indexOf("*/");
                if (e != -1) { inBlock = false; col = e + 2; }
                else continue;
            }

            while (col < line.length()) {
                char ch = line.charAt(col);

                // 跳过空白
                if (Character.isWhitespace(ch)) { col++; continue; }

                // 单行注释 //
                if (ch == '/' && col + 1 < line.length() && line.charAt(col + 1) == '/') break;

                // 多行注释 /* */
                if (ch == '/' && col + 1 < line.length() && line.charAt(col + 1) == '*') {
                    int e = line.indexOf("*/", col + 2);
                    if (e != -1) { col = e + 2; continue; }
                    else { inBlock = true; break; }
                }

                // 双字符运算符 :=, <=, >=
                if (col + 1 < line.length()) {
                    String t = line.substring(col, col + 2);
                    if (":=".equals(t) || "<=".equals(t) || ">=".equals(t)) {
                        toks.add(new Token("Op", t, lineno));
                        col += 2; continue;
                    }
                }

                // 单字符运算符
                if ("+-*/=#<>".indexOf(ch) != -1) {
                    toks.add(new Token("Op", "" + ch, lineno));
                    col++; continue;
                }

                // 界符
                if ("();,.".indexOf(ch) != -1) {
                    toks.add(new Token("Delim", "" + ch, lineno));
                    col++; continue;
                }

                // 标识符或关键字
                if (Character.isLetter(ch) || ch == '_') {
                    int s = col;
                    while (col < line.length()
                        && (Character.isLetterOrDigit(line.charAt(col)) || line.charAt(col) == '_'))
                        col++;
                    String w = line.substring(s, col);
                    if (KW.contains(w))
                        toks.add(new Token("KW", w, lineno));
                    else {
                        if (w.length() > 8)
                            errs.add(new LexerError("LongIdent," + w, lineno));
                        toks.add(new Token("Ident", w, lineno));
                    }
                    continue;
                }

                // 数字
                if (Character.isDigit(ch)) {
                    int s = col;
                    while (col < line.length() && Character.isDigit(line.charAt(col))) col++;
                    String n = line.substring(s, col);

                    // 数字后跟字母 → 非法单词, 如 2a
                    if (col < line.length() && Character.isLetter(line.charAt(col))) {
                        while (col < line.length()
                            && (Character.isLetterOrDigit(line.charAt(col)) || line.charAt(col) == '_'))
                            col++;
                        String b = line.substring(s, col);
                        errs.add(new LexerError("BadToken," + b, lineno));
                        toks.add(new Token("BadToken", b, lineno));
                        continue;
                    }

                    if (n.length() > 8)
                        errs.add(new LexerError("NumOverflow," + n, lineno));
                    toks.add(new Token("Num", n, lineno));
                    continue;
                }

                // 非法字符 (@ & ! 等)
                errs.add(new LexerError("BadChar," + ch, lineno));
                col++;
            }
        }
        return toks;
    }
}
```

```java
public class Quadruple {
    public String op;      // +, -, *, /, :=, j, j<, ...
    public String arg1;    // 第一操作数
    public String arg2;    // 第二操作数
    public String result;  // 结果 / 跳转目标
}
```

### 四元式管理器（含回填）

```java
// src/quadruple/QuadrupleManager.java
public class QuadrupleManager {
    private List<Quadruple> quads = new ArrayList<>();
    private int tempCount = 0;                    // 临时变量计数器 T1, T2...
    private int labelCount = 0;                   // 标签计数器 $1, $2...
    private Map<String, List<Integer>> pending = new HashMap<>(); // 待回填表

    // 发射一条四元式
    public int emit(String op, String a1, String a2, String r) {
        quads.add(new Quadruple(op, a1, a2, r));
        return quads.size();  // 返回 1-based 序号
    }

    // 分配新临时变量
    public String newTemp() {
        tempCount++;
        return "T" + tempCount;
    }

    // 创建新标签（回填用）
    public String newLabel() {
        labelCount++;
        return "$" + labelCount;
    }

    // 带待回填的发射：先发射，再将序号加入 pending 表
    public int emitWP(String op, String a1, String a2, String label) {
        int idx = emit(op, a1, a2, label);
        pending.computeIfAbsent(label, k -> new ArrayList<>()).add(idx);
        return idx;
    }

    // 回填：将此前所有 emitWP 的四元式 result 更新为当前位置
    public int patch(String label) {
        int pos = quads.size() + 1;
        List<Integer> idxs = pending.get(label);
        if (idxs != null) {
            for (int i : idxs)
                quads.get(i - 1).result = "" + pos;
            pending.remove(label);
        }
        return pos;
    }

    // 输出全部四元式
    public String toStr() {
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < quads.size(); i++)
            sb.append("(").append(i + 1).append(")").append(quads.get(i)).append("\n");
        return sb.toString().trim();
    }
}
```

## 3. 符号表核心 — SymbolTable.java

```java
// src/symbol/Symbol.java
public class Symbol {
    public String name;   // 标识符名
    public String kind;   // const / var / procedure
    public String value;  // 常量值 (变量/过程为 "0")
    public int level;     // 作用域层级
    public int addr;      // 地址(预留)
}

// src/symbol/SymbolTable.java
public class SymbolTable {
    // 作用域栈: 每个作用域一个 HashMap
    private List<Map<String, Symbol>> scopes = new ArrayList<>();
    private List<Symbol> all = new ArrayList<>();  // 全部符号

    public SymbolTable() { enterScope(); }

    // 进入新作用域
    public void enterScope() {
        scopes.add(new HashMap<>());
    }

    // 退出当前作用域
    public void exitScope() {
        if (scopes.size() > 1)
            scopes.remove(scopes.size() - 1);
    }

    public int level() { return scopes.size() - 1; }

    // 在当前作用域添加符号（重复声明返回 false）
    public boolean add(String name, String kind, String value) {
        Map<String, Symbol> cur = scopes.get(scopes.size() - 1);
        if (cur.containsKey(name)) return false;
        Symbol s = new Symbol(name, kind, value, level(), 0);
        cur.put(name, s);
        all.add(s);
        return true;
    }

    // 从内向外逐层查找符号
    public Symbol lookup(String name) {
        for (int i = scopes.size() - 1; i >= 0; i--) {
            Map<String, Symbol> s = scopes.get(i);
            if (s.containsKey(name)) return s.get(name);
        }
        return null;
    }
}
```

### Token 类型映射 (sym 函数)

```java
private String sym(Token t) {
    if (t == null) return "EOF";
    switch (t.value) {
        case "const":     return "CONST";
        case "var":       return "VAR";
        case "procedure": return "PROC";
        case "call":      return "CALL";
        case "begin":     return "BEG";
        case "end":       return "END";
        case "if":        return "IF";
        case "then":      return "THEN";
        case "else":      return "ELSE";
        case "while":     return "WHILE";
        case "do":        return "DO";
        case "read":      return "READ";
        case "write":     return "WRITE";
        case "odd":       return "ODD";
        case ":=":        return "ASGN";
        case "=":         return "EQ";
        case "#":         return "NE";
        case "<":         return "LT";
        case "<=":        return "LE";
        case ">":         return "GT";
        case ">=":        return "GE";
        case "+":         return "PLUS";
        case "-":         return "MIN";
        case "*":         return "MUL";
        case "/":         return "DIV";
        case "(":         return "LP";
        case ")":         return "RP";
        case ";":         return "SEMI";
        case ",":         return "COM";
        case ".":         return "DOT";
    }
    if ("Ident".equals(t.kind)) return "id";
    if ("Num".equals(t.kind))   return "num";
    return t.kind;
}
```

### 主入口 — parse 方法

```java
public Map<String, Object> parse(List<Token> tks, List<LexerError> le) {
    toks = tks;
    ti = 0;
    errs.clear();
    qm = new QuadrupleManager();
    st = new SymbolTable();

    qm.emit("syss");         // 程序开始标记

    parseBlock();            // 解析块
    if (!mat("DOT")) {
        Token t = peek();
        err("Expect . got " + (t != null ? t.value : "EOF"),
            t != null ? t.lineno : 0);
    }

    qm.emit("syse");         // 程序结束标记

    Map<String, Object> r = new HashMap<>();
    r.put("ok", errs.isEmpty());
    r.put("quads", qm.toStr());
    r.put("symtbl", st.toString());
    r.put("errs", errs);
    r.put("lexerrs", le != null ? le : new ArrayList<>());
    return r;
}
```

### 常量声明

```java
// const id = num ; [const ...]
private void parseConst() {
    if (!mat("CONST")) return;

    Token id = requireId();      // 取标识符
    String n = id.value;
    int ln = id.lineno;

    require("EQ", "Expect =");   // 匹配 =
    Token nu = requireNum();     // 取数字

    require("SEMI", "Expect ;"); // 匹配 ;

    if (!st.add(n, "const", nu.value))
        err("Dup const " + n, ln);
    else {
        qm.emit("const", n);
        qm.emit("=", nu.value, "_", n);
    }

    parseConst();  // 递归处理连续常量声明
}
```

### 变量声明

```java
// var id {, id} ;
private void parseVar() {

private void parseIds() {
    Token id = requireId();
    String n = id.value;
    int ln = id.lineno;

    if (!st.add(n, "var"))
        err("Dup var " + n, ln);
    else
        qm.emit("var", n);

    if (mat("COM")) parseIds();  // 递归处理逗号分隔
}
```

### 过程声明

```java
// procedure id ; block ;
private void parseProc() {
    if (!mat("PROC")) return;
    Token id = requireId();
    int ln = id.lineno;

    require("SEMI", "Expect ;");

    if (!st.add(id.value, "procedure"))
        err("Dup proc " + id.value, ln);
    qm.emit("procedure", id.value);

    st.enterScope();     // ← 进入新作用域
    parseBlock();
    st.exitScope();      // ← 退出作用域

    require("SEMI", "Expect ; after proc");
    parseProc();         // 递归处理连续过程声明
}
```

### 语句分发

```java
private void parseStmt() {
    Token t = peek();
    if (t == null) return;

    switch (sym(t)) {
        case "id":    parseAssign(); break;  // 赋值
        case "READ":  parseRead();   break;  // 读
        case "WRITE": parseWrite();  break;  // 写
        case "CALL":  parseCall();   break;  // 调用
        case "BEG":                    // 复合语句 begin...end
            next();
            parseStmtList();
            require("END", "Expect END");
            break;
        case "IF":    parseIf();    break;   // 条件
        case "WHILE": parseWhile(); break;   // 循环
        case "ODD":   parseCond();  break;   // 条件 (odd)
        default: break;                      // 空语句
    }
}
```

```java
// id := expr ;
private void parseAssign() {
    Token id = next();
    String n = id.value;
    int ln = id.lineno;

    Symbol s = st.lookup(n);
    if (s == null)
        err("Undecl " + n, ln);
    else if (!"var".equals(s.kind))
        err(n + " not var", ln);

    require("ASGN", "Expect :=");
    String ep = parseExpr();
    require("SEMI", "Expect ;");

    qm.emit(":=", ep, "_", n);
}
```

### 条件语句 if-then-else（含回填）

```java
// if cond then stmt [else stmt]
private void parseIf() {
    next();  // 消费 if

    String fl = qm.newLabel();   // false → 跳转标签
    String el = qm.newLabel();   // end   → 结束时标签

    String cp = parseCond();     // 解析条件

    require("THEN", "Expect THEN");

    // 条件为假时跳转到 fl
    qm.emitWP("j=", cp, "0", fl);

    parseStmt();                 // 解析 then 子句

    // then 结束后跳过 else 部分
    qm.emitWP("j", "_", "_", el);
    qm.patch(fl);                // 回填 fl → 当前位置

    if (mat("ELSE"))
        parseStmt();             // 解析 else 子句（可选）

    qm.patch(el);                // 回填 el → 当前位置
}
```

### 循环语句 while-do（含回填）

```java
// while cond do stmt
private void parseWhile() {
    next();  // 消费 while

    String ls = qm.newLabel();
    int loopStart = qm.patch(ls);     // ← 记录循环开始位置

    String ex = qm.newLabel();        // 出口标签

    String cp = parseCond();          // 解析条件

    require("DO", "Expect DO");

    qm.emitWP("j=", cp, "0", ex);    // 条件为假 → 跳出
    parseStmt();                      // 循环体

    qm.emit("j", "_", "_", String.valueOf(loopStart));  // 跳回条件
    qm.patch(ex);                     // 回填出口
}
```

### 表达式解析（消除左递归）

```java
// 表达式 → 项 {(+|-) 项}
private String parseExpr() {
    return parseETail(parseTerm());
}

// 尾递归处理加减法
private String parseETail(String l) {
    Token t = peek();
    if (t == null) return l;

    String s = sym(t);
    if (s.equals("PLUS") || s.equals("MIN")) {
        next();
        String rr = parseTerm();
        String r = qm.newTemp();
        qm.emit(s.equals("PLUS") ? "+" : "-", l, rr, r);
        return parseETail(r);  // 尾递归
    }
    return l;
}

// 项 → 因子 {(*|/) 因子}
private String parseTerm() {
    return parseTTail(parseFact());
}

private String parseTTail(String l) {
    Token t = peek();
    if (t == null) return l;

    String s = sym(t);
    if (s.equals("MUL") || s.equals("DIV")) {
        next();
        String rr = parseFact();
        String r = qm.newTemp();
        qm.emit(s.equals("MUL") ? "*" : "/", l, rr, r);
        return parseTTail(r);  // 尾递归
    }
    return l;
}

// 因子 → id | num | ( expr )
private String parseFact() {
    Token t = peek();
    if (t == null) { err("Unexpected EOF"); return "_"; }

    String s = sym(t);

    if (s.equals("id")) {         // 标识符
        Token id = next();
        Symbol ss = st.lookup(id.value);
        if (ss == null) {
            err("Undecl " + id.value, id.lineno);
            return "_";
        }
        return id.value;
    }

    if (s.equals("num"))          // 数字
        return next().value;

    if (s.equals("LP")) {         // 子表达式 ( expr )
        next();
        String e = parseExpr();
        if (!mat("RP")) err("Expect )");
        return e;
    }

    err("Bad expr start " + t.value, t.lineno);
    return "_";
}
```

### 条件解析

```java
// 条件 → expr relop expr | odd expr
private String parseCond() {
    Token t = peek();
    if (t == null) return "_";

    // odd expr: 判断奇偶
    if (sym(t).equals("ODD")) {
        next();
        String e = parseExpr();
        String r = qm.newTemp();
        qm.emit("%", e, "2", r);  // T = expr % 2
        return r;
    }

    // expr relop expr
    String l = parseExpr();
    Token ot = peek();
    if (ot != null) {
        String rp = sym(ot);
        if (rp.equals("EQ") || rp.equals("NE") || rp.equals("LT") ||
            rp.equals("LE") || rp.equals("GT") || rp.equals("GE")) {
            next();
            String op = Map.of(
                "EQ", "=", "NE", "#", "LT", "<",
                "LE", "<=", "GT", ">", "GE", ">="
            ).get(rp);
            String rr = parseExpr();
            String r = qm.newTemp();
            qm.emit(op, l, rr, r);  // T = l op rr
            return r;
        }
    }
    return l;
}
```

```java
// src/main/Main.java
public class Main {
    public static void main(String[] args) throws Exception {
        if (args.length == 0) runExamples();       // 示例模式
        else {
            File f = new File(args[0]);
            if (f.isDirectory()) batch(f);         // 批量模式
            else file(f);                          // 单文件模式
        }
    }

    // 单文件处理
    static void file(File f) throws Exception {
        String code = new String(Files.readAllBytes(f.toPath()), "UTF-8");

        Lexer lexer = new Lexer();
        List<Token> tokens = lexer.tokenize(code);
        List<LexerError> lexErrs = lexer.errs;

        LR1Parser parser = new LR1Parser();
        Map<String, Object> result = parser.parse(tokens, lexErrs);

        // 输出结果...
        boolean ok = (boolean) result.get("ok");
        if (ok) {
            System.out.println("语义正确");
            System.out.println("\n中间代码:\n" + result.get("quads"));
            System.out.println("符号表:\n" + result.get("symtbl"));
        } else {
            for (SemanticError e : (List<SemanticError>) result.get("errs"))
                System.out.println(e);
        }
    }
}
```
